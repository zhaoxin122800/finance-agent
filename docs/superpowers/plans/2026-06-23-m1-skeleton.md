**Goal:** 搭好 finance-agent 的项目骨架 —— 复制全部 vendors 资产、实现带缓存的 `DataSource` 抽象层、给出能跑的首页和导航壳子,使 `uv run python -m app.main` 起服务后浏览器能打开 `http://127.0.0.1:8000` 看到带 8 个导航 tab 的空页面。

`app/datasources/free_source.py`:

```python
"""FreeSource —— 默认数据源实现,转调 vendors.data + 内存缓存 + 错误兜底。

缓存只对 realtime 生效(TTL 防新浪封 IP);其它接口每次实时调,
因为它们调用频率低且数据量大不宜常驻。
"""
from __future__ import annotations

import logging
import time
from typing import Any

from vendors import data  # 原样资产,作为依赖注入点

log = logging.getLogger("finance-agent.datasource")


class FreeSource:
    """默认免费数据源。"""

    def __init__(self, cache_ttl_realtime: int = 30):
        self._ttl = cache_ttl_realtime
        # 缓存结构: {frozenset(sorted_codes): (timestamp, payload)}
        self._realtime_cache: dict[frozenset[str], tuple[float, dict[str, dict]]] = {}

    def get_realtime(self, codes: list[str]) -> dict[str, dict]:
        if not codes:
            return {}
        key = frozenset(codes)
        now = time.time()
        cached = self._realtime_cache.get(key)
        if cached and (now - cached[0]) < self._ttl:
            return cached[1]
        try:
            payload = data.get_realtime(list(codes))
        except Exception as e:
            log.warning("get_realtime 失败(%s): %s", codes, e)
            return {}
        self._realtime_cache[key] = (now, payload)
        return payload

    def get_kline(self, code: str, days: int = 30, scale: str = "240") -> list[dict]:
        try:
            return data.get_kline(code, days, scale)
        except Exception as e:
            log.warning("get_kline 失败(%s): %s", code, e)
            return []

    def get_minute(self, code: str) -> list[dict]:
        try:
            return data.get_minute(code)
        except Exception as e:
            log.warning("get_minute 失败(%s): %s", code, e)
            return []

    def get_sectors(self) -> list[dict]:
        try:
            return data.get_sectors()
        except Exception as e:
            log.warning("get_sectors 失败: %s", e)
            return []

    def get_money_flow(self, code: str) -> dict:
        try:
            return data.get_money_flow(code)
        except Exception as e:
            log.warning("get_money_flow 失败(%s): %s", code, e)
            return {}

    def get_top_list(self, date: str | None = None) -> list[dict]:
        try:
            return data.get_top_list(date)
        except Exception as e:
            log.warning("get_top_list 失败: %s", e)
            return []
```

"""data/ 运行时 JSON 状态读写 + schema 校验。

设计 §8:状态文件损坏时备份后重置为安全默认值,不崩。
"""
from __future__ import annotations

import json
import logging
import time
from pathlib import Path
from typing import Any

from app.config import DATA_DIR

log = logging.getLogger("finance-agent.state")


def load_json(filename: str, default: Any = None) -> Any:
    """读 data/<filename>。损坏则备份后返回 default。"""
    path = DATA_DIR / filename
    if not path.exists():
        return default
    try:
        with path.open("r", encoding="utf-8") as f:
            return json.load(f)
    except (json.JSONDecodeError, OSError) as e:
        log.warning("状态文件损坏 %s: %s —— 备份后返回默认值", path, e)
        backup = path.with_suffix(path.suffix + f".corrupt.{int(time.time())}")
        try:
            path.rename(backup)
        except OSError:
            pass
        return default


def save_json(filename: str, payload: Any) -> None:
    """写 data/<filename>,自动建父目录。"""
    path = DATA_DIR / filename
    path.parent.mkdir(parents=True, exist_ok=True)
    tmp = path.with_suffix(path.suffix + ".tmp")
    with tmp.open("w", encoding="utf-8") as f:
        json.dump(payload, f, ensure_ascii=False, indent=2)
    tmp.replace(path)  # 原子替换


def validate_sim_account(path: Path) -> bool:
    """校验 sim_account.json 的最小 schema。"""
    required = {"initial_capital", "cash", "positions", "history"}
    try:
        with path.open("r", encoding="utf-8") as f:
            data = json.load(f)
    except (json.JSONDecodeError, OSError):
        return False
    if not isinstance(data, dict):
        return False
    return required.issubset(data.keys()) and isinstance(data["positions"], dict) and isinstance(data["history"], list)
```

"""启动时确保 sim_account.json 存在且有效。

设计 §8:启动时校验 schema,损坏则备份后重置为初始 1000 元。
"""
from __future__ import annotations

import logging
from datetime import datetime, timezone
from pathlib import Path

from app.services.state import save_json, validate_sim_account

log = logging.getLogger("finance-agent.account_seed")

INITIAL_CAPITAL = 1000.0


def _fresh_account() -> dict:
    now = datetime.now(timezone.utc).isoformat()
    return {
        "initial_capital": INITIAL_CAPITAL,
        "cash": INITIAL_CAPITAL,
        "positions": {},
        "history": [],
        "created_at": now,
        "updated_at": now,
    }


def ensure_sim_account(path: str | Path) -> None:
    """path 是 data/sim_account.json 的绝对路径。不存在或无效则(重)置。"""
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    if p.exists() and validate_sim_account(p):
        return
    if p.exists():
        log.warning("sim_account.json 无效,重置为初始 %s 元", INITIAL_CAPITAL)
    save_json(Path(path).name, _fresh_account())
```

> **注意**:`save_json` 的 filename 参数是相对 DATA_DIR 的,这里传 `Path(path).name` 取文件名部分。测试 fixture 把 DATA_DIR 重定向到 tmp_path,所以写入位置正确。

def create_app() -> FastAPI:

        StaticFiles(directory=str(PROJECT_ROOT / "app" / "static")),
